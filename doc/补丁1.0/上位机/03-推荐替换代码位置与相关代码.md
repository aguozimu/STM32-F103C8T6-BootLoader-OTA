### 3.3 `renderDevice()` 推荐最终代码

```javascript
function renderDevice() {
  const device = appState.device;
  const statusName = device.run_status_name || "UNKNOWN";
  const statusText = device.run_status_zh || STATUS_ZH[statusName] || "未查询";
  const statusNode = $("runStatus");
  statusNode.textContent = statusText;
  statusNode.className = "";

  if (device.run_status === 0x05 ||
      device.run_status === 0x06 ||
      device.run_status === 0x09) {
    statusNode.classList.add("ok");
  } else if (device.run_status === 0x03 ||
             device.run_status === 0x04 ||
             device.run_status === 0x08) {
    statusNode.classList.add("warn");
  } else if (device.run_status === 0x07 ||
             device.run_status === 0x0A) {
    statusNode.classList.add("bad");
  }

  $("runStatusRaw").textContent =
    device.run_status === null ? "UNKNOWN" : statusName;
  $("programId").textContent = formatId(device.program_id);
  $("currentPageBlock").textContent =
    `${device.page === null ? "—" : `0x${hex(device.page, 2)}`} / ` +
    `${device.block === null ? "—" : `0x${hex(device.block, 2)}`}`;

  const errorNode = $("errorCode");
  errorNode.textContent = `0x${hex(device.error_code || 0, 2)}`;
  errorNode.className = `mono${device.error_code ? " bad" : ""}`;
  $("errorDesc").textContent =
    device.error_code ? (device.error_zh || device.error_name) : "无错误";
  $("lastQuery").textContent = device.last_query_at
    ? `最近查询：${formatTime(device.last_query_at)}` +
      `${device.last_query_ms === null ? "" : ` · ${device.last_query_ms} ms`}`
    : "尚未进行设备查询";

  const badge = $("deviceQueryBadge");
  // 保持文件中原有后续代码不变。
}
```

### 3.4 不允许的修改

不要将以下任一状态加入 OTA 下载成功条件：

- `0x06 RUNNING`
- `0x08 RUN_PENDING_CONFIRM`
- `0x09 RUN_CONFIRMED_OK`
- `0x0A RUN_CONFIRMED_BAD`

现有最终成功判断必须保留：

```python
if final_sta is None or final_sta["run_status"] != 0x05 or final_sta["error_code"] != 0:
    raise OtaError("最终 SetSta 未报告 VALID_BACKUP")
if final_end["run_status"] != 0x05 or final_end["error_code"] != 0:
    raise OtaError("最终 SetEnd 未报告成功")
```

```python
def test_extended_run_status_names(self):
    expected = {
        0x08: ("RUN_PENDING_CONFIRM", "运行待确认"),
        0x09: ("RUN_CONFIRMED_OK", "确认运行正常"),
        0x0A: ("RUN_CONFIRMED_BAD", "确认无法运行"),

    for code, (name, zh) in expected.items():
        parsed = parse_status(status_record(code, 0xFF, 0xFF, 0x00))
        self.assertEqual(parsed["run_status"], code)
        self.assertEqual(parsed["run_status_name"], name)
        self.assertEqual(parsed["run_status_zh"], zh)
        self.assertEqual(parsed["page"], 0xFF)
        self.assertEqual(parsed["block"], 0xFF)
        self.assertEqual(parsed["error_code"], 0x00)
```

```python
def test_query_status_accepts_extended_states(self):
    cases = (
        (0x08, "RUN_PENDING_CONFIRM", "运行待确认"),
        (0x09, "RUN_CONFIRMED_OK", "确认运行正常"),
        (0x0A, "RUN_CONFIRMED_BAD", "确认无法运行"),
    )

    for code, name, zh in cases:
        with self.subTest(code=code):
            fake = FakeDeviceLink(
                [0] * 16,
                [0] * 16,
                program_id=0x0003,
                query_status=status_record(code, 0xFF, 0xFF, 0x00),
            )
            engine = OtaEngine(fake)
            engine.start_query("status")
            self.assertTrue(engine.wait_idle(2.0))
            device = engine.snapshot(include_logs=False)["device"]
            self.assertEqual(device["run_status"], code)
            self.assertEqual(device["run_status_name"], name)
            self.assertEqual(device["run_status_zh"], zh)
```
