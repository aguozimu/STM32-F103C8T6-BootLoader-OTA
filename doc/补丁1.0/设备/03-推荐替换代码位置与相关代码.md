### 3.1 新增控制页读取与校验

位置：`FlashManage_BufferCRC()`、`FlashManage_FlashPageCRC()` 之后。

```c
static uint8_t FlashManage_ReadControlPage(uint32_t control_page, Data_Backup* data)
{
    uint32_t stored_crc;
    uint8_t page;

    MyFLASH_ReadBuffer(control_page, (uint32_t*)data, 33U);
    if ((data->ID == 0U) || (data->Run_status > OTA_RUN_STATUS_RUN_CONFIRMED_BAD)) {
        return 0U;
    }

    stored_crc = data->Page[15].Page_CRC;
    MyFLASH_ReadBuffer(control_page, FlashManage_PageBuffer.words, OTA_PAGE_WORD_COUNT);
    FlashManage_WriteU32(&FlashManage_PageBuffer.bytes[0x80], 0U);
    if (FlashManage_BufferCRC(FlashManage_PageBuffer.words) != stored_crc) {
        return 0U;
    }

    for (page = 0U; page < OTA_PROGRAM_PAGE_COUNT; page++) {
        uint32_t address = (data->Page[page].Page_ADD < 0x4000U)
            ? APP_RUN_BASE + data->Page[page].Page_ADD
            : data->Page[page].Page_ADD;
        if (FlashManage_FlashPageCRC(address) != data->Page[page].Page_CRC) {
            return 0U;
        }
    }
    return 1U;
}
```

> 这里对绝对/相对地址都做兼容，便于读取一号槽和缓存槽。正式实现应增加显式地址范围检查。

### 3.2 替换 `FlashManage_FirstOnly()`

位置：约第 65~95 行。推荐替换：

```c
void FlashManage_InitIfNeeded(void)
{
    Data_Backup running;

    if (!FlashManage_ReadControlPage(BackupPageADD1, &running)) {
        /* 仅控制页无效时重建，禁止覆盖 BootLoader 写入的 ID/状态 */
        Data_Backup rebuilt = {0};
        uint8_t page;

        rebuilt.ID = APP_MID;
        rebuilt.Run_status = OTA_RUN_STATUS_RUNNING;
        rebuilt.Data_area = OTA_DATA_AREA_COMPLETE;
        for (page = 0U; page < OTA_PAGE_COUNT; page++) {
            rebuilt.Page[page].Page_ADD = (uint32_t)page * OTA_PAGE_SIZE;
            rebuilt.Page[page].Page_CRC = (page < OTA_PROGRAM_PAGE_COUNT)
                ? FlashManage_FlashPageCRC(APP_RUN_BASE + ((uint32_t)page * OTA_PAGE_SIZE))
                : 0U;

        FlashManage_WriteControlPage(BackupPageADD1, &rebuilt,
```

### 3.3 查找匹配源槽

位置：`FlashManage_SelectTargetSlot()` 附近新增：

```c
static uint8_t FlashManage_FindMatchingSourceSlot(
    const Data_Backup* running,
    uint8_t* slot_index)
{
    uint8_t found = 0xFFU;
    uint8_t candidate;
    uint8_t page;

    for (candidate = 0U; candidate < 2U; candidate++) {
        Data_Backup source;
        uint32_t control_page = FlashManage_Slots[candidate].control_page;

        if (!FlashManage_ReadControlPage(control_page, &source)) {
            continue;
        }
        if (source.ID != running->ID) {
            continue;
        }
        if ((source.Run_status != OTA_RUN_STATUS_RUN_PENDING_CONFIRM) &&
            (source.Run_status != OTA_RUN_STATUS_RUN_CONFIRMED_OK)) {
            continue;
        }
        for (page = 0U; page < OTA_PROGRAM_PAGE_COUNT; page++) {
            if (source.Page[page].Page_CRC != running->Page[page].Page_CRC) {
                break;
            }
        }
        if (page != OTA_PROGRAM_PAGE_COUNT) {
            continue;
        }
        if (found != 0xFFU) {
            return 0xFFU; /* 多个匹配时不猜测 */
        }
        found = candidate;
    }

    if (found != 0xFFU) {
        *slot_index = found;
    }
    return found;
}
```

### 3.4 自检成功接口

```c
void FlashManage_ConfirmRunning(void)
{
    Data_Backup running;
    Data_Backup source;
    uint8_t source_slot = 0xFFU;
    uint8_t other_slot;

    if (!FlashManage_ReadControlPage(BackupPageADD1, &running)) {
        return;
    }

    (void)FlashManage_FindMatchingSourceSlot(&running, &source_slot);
    if (running.Run_status != OTA_RUN_STATUS_RUNNING) {
        FlashManage_WriteControlPage(BackupPageADD1, &running,
            OTA_RUN_STATUS_RUNNING, OTA_DATA_AREA_COMPLETE);
    }

    if (source_slot != 0xFFU) {
        uint32_t source_control = FlashManage_Slots[source_slot].control_page;
        if (FlashManage_ReadControlPage(source_control, &source) &&
            (source.Run_status != OTA_RUN_STATUS_RUN_CONFIRMED_OK)) {
            FlashManage_WriteControlPage(source_control, &source,
                OTA_RUN_STATUS_RUN_CONFIRMED_OK, OTA_DATA_AREA_COMPLETE);
        }

        other_slot = (source_slot == 0U) ? 1U : 0U;
        {
            uint32_t other_control = FlashManage_Slots[other_slot].control_page;
            Data_Backup other;
            if (FlashManage_ReadControlPage(other_control, &other) &&
                (other.Run_status != OTA_RUN_STATUS_RUN_CONFIRMED_BAD) &&
                (other.Run_status != OTA_RUN_STATUS_VALID_BACKUP) &&
                (other.Run_status != OTA_RUN_STATUS_RECEIVING) &&
                (other.Run_status != OTA_RUN_STATUS_PAUSED) &&
                (other.Run_status != OTA_RUN_STATUS_VERIFYING)) {
                FlashManage_WriteControlPage(other_control, &other,
                    OTA_RUN_STATUS_EMPTY, OTA_DATA_AREA_COMPLETE);
            }
        }
    }
}
```

该函数是幂等的：一号槽已是 `0x06`、源槽已是 `0x09` 时不会重复擦写。

### 3.5 自检失败接口

```c
void FlashManage_RejectRunning(void)
{
    Data_Backup running;
    Data_Backup source;
    uint8_t source_slot = 0xFFU;

    if (FlashManage_ReadControlPage(BackupPageADD1, &running)) {
        FlashManage_WriteControlPage(BackupPageADD1, &running,
            OTA_RUN_STATUS_RUN_CONFIRMED_BAD, OTA_DATA_AREA_COMPLETE);

        if (FlashManage_FindMatchingSourceSlot(&running, &source_slot) != 0xFFU) {
            uint32_t source_control = FlashManage_Slots[source_slot].control_page;
            if (FlashManage_ReadControlPage(source_control, &source)) {
                FlashManage_WriteControlPage(source_control, &source,
                    OTA_RUN_STATUS_RUN_CONFIRMED_BAD, OTA_DATA_AREA_COMPLETE);
            }
        }
    }

    NVIC_SystemReset();
}
```

注意：匹配源槽必须在把 `running.Run_status` 改成 `0x0A` 之前使用原始元数据完成查找。若匹配函数依赖状态，应先保存源槽号再写状态。

### 3.6 保持 OTA 最终握手不变

`FlashManage_HandleSetEnd()` 和 `FlashManage_SendFinalSuccess()` 仍使用 `OTA_RUN_STATUS_VALID_BACKUP`。不要把 OTA 最终成功改成 `0x06/0x08/0x09/0x0A`。

`GetSta` 仍读取 `BackupPageADD1`，因此上位机只会通过 `GetSta` 看到当前一号槽状态。源槽的 `0x09/0x0A` 是设备内部回滚依据，不需要新增查询命令。

```c
void MyFLASH_ReadBuffer(uint32_t Address, uint32_t* Data, uint16_t WordCount)
{
    uint16_t index;
    for (index = 0U; index < WordCount; index++) {
        Data[index] = *((__IO uint32_t*)(Address + (index * 4U)));
    }
}

void MyFLASH_ProgramBuffer(uint32_t Address, const uint32_t* Data, uint16_t WordCount)
{
    uint16_t index;
    HAL_FLASH_Unlock();
    for (index = 0U; index < WordCount; index++) {
        (void)HAL_FLASH_Program(FLASH_TYPEPROGRAM_WORD,
            Address + (index * 4U), (uint64_t)Data[index]);
    }
    HAL_FLASH_Lock();
}
```

```c
    {0x0800BC00U, 0x08008000U},
```

### 7.2 CRC 与元数据校验

```c
static uint32_t OtaBoot_PageCrc(uint32_t address)
{
    return HAL_CRC_Calculate(&hcrc, (uint32_t*)address,
        OTA_BOOT_PAGE_SIZE / 4U);
}

static uint8_t OtaBoot_ReadMetadata(OtaBoot_Slot slot, OtaBoot_Metadata* metadata)
{
    MyFLASH_ReadBuffer(OtaBoot_Slots[slot].control_page,
        (uint32_t*)metadata, sizeof(OtaBoot_Metadata) / 4U);

    if (metadata->ID == 0U) {
        return 0U;
    }
    if (metadata->Run_status > OTA_BOOT_STATUS_RUN_CONFIRMED_BAD) {
        return 0U;
    }
    return 1U;
}

static uint8_t OtaBoot_ValidateControlPage(
    OtaBoot_Slot slot,
    OtaBoot_Metadata* metadata)
{
    uint32_t stored_crc;

    if (!OtaBoot_ReadMetadata(slot, metadata)) {
        return 0U;
    }

    stored_crc = metadata->Page[15].Page_CRC;
    MyFLASH_ReadBuffer(OtaBoot_Slots[slot].control_page,
        OtaBoot_PageBuffer, OTA_BOOT_PAGE_SIZE / 4U);
    OtaBoot_PageBuffer[0x80U / 4U] = 0U;

    if (HAL_CRC_Calculate(&hcrc, OtaBoot_PageBuffer,
            OTA_BOOT_PAGE_SIZE / 4U) != stored_crc) {
        return 0U;
    }
    return 1U;
}

static uint8_t OtaBoot_ValidateProgramPages(
    OtaBoot_Slot slot,
    const OtaBoot_Metadata* metadata)
{
    uint8_t page;
    for (page = 0U; page < OTA_BOOT_PROGRAM_PAGE_COUNT; page++) {
        uint32_t address = OtaBoot_Slots[slot].data_base +
            ((uint32_t)page * OTA_BOOT_PAGE_SIZE);
        if (OtaBoot_PageCrc(address) != metadata->Page[page].Page_CRC) {
            return 0U;
        }
    }
    return 1U;
}

static uint8_t OtaBoot_ValidateSlot(
    OtaBoot_Slot slot,
    OtaBoot_Metadata* metadata)
{
    return OtaBoot_ValidateControlPage(slot, metadata) &&
           OtaBoot_ValidateProgramPages(slot, metadata);
}
```

### 7.3 写控制页

```c
static uint8_t OtaBoot_WriteMetadata(
    OtaBoot_Slot slot,
    const OtaBoot_Metadata* source,
    uint8_t status)
{
    OtaBoot_Metadata metadata = *source;
    uint32_t control_page = OtaBoot_Slots[slot].control_page;
    uint32_t control_crc;
    uint8_t page;

    metadata.Run_status = status;
    metadata.Data_area = OTA_BOOT_DATA_AREA_COMPLETE;
    for (page = 0U; page < OTA_BOOT_PAGE_COUNT; page++) {
        metadata.Page[page].Page_ADD = (slot == OTA_BOOT_SLOT_1)
            ? ((uint32_t)page * OTA_BOOT_PAGE_SIZE)
            : (OtaBoot_Slots[slot].data_base + ((uint32_t)page * OTA_BOOT_PAGE_SIZE));
    }
    metadata.Page[15].Page_ADD = control_page;
    metadata.Page[15].Page_CRC = 0U;

    for (page = 0U; page < OTA_BOOT_PAGE_SIZE / 4U; page++) {
        OtaBoot_PageBuffer[page] = 0xFFFFFFFFU;
    }
    memcpy(OtaBoot_PageBuffer, &metadata, sizeof(metadata));

    control_crc = HAL_CRC_Calculate(&hcrc, OtaBoot_PageBuffer,
        OTA_BOOT_PAGE_SIZE / 4U);
    ((OtaBoot_Metadata*)OtaBoot_PageBuffer)->Page[15].Page_CRC = control_crc;

    MyFLASH_ErasePage(control_page);
    MyFLASH_ProgramBuffer(control_page, OtaBoot_PageBuffer,
        OTA_BOOT_PAGE_SIZE / 4U);

    return (OtaBoot_PageCrc(control_page) == control_crc) ? 1U : 0U;
}
```

> 设备写回控制页时统一保存对应槽的实际绝对地址；上位机发送控制页时才使用 `0x0000~0x3C00` 的相对地址。不要在设备最终元数据中混用两种地址规则。

### 7.4 复制程序页

```c
static uint8_t OtaBoot_CopyProgramPages(
    OtaBoot_Slot source_slot,
    const OtaBoot_Metadata* metadata)
{
    uint8_t page;

    for (page = 0U; page < OTA_BOOT_PROGRAM_PAGE_COUNT; page++) {
        uint32_t source_address = OtaBoot_Slots[source_slot].data_base +
            ((uint32_t)page * OTA_BOOT_PAGE_SIZE);
        uint32_t target_address = OtaBoot_Slots[OTA_BOOT_SLOT_1].data_base +
            ((uint32_t)page * OTA_BOOT_PAGE_SIZE);

        MyFLASH_ReadBuffer(source_address, OtaBoot_PageBuffer,
            OTA_BOOT_PAGE_SIZE / 4U);
        if (HAL_CRC_Calculate(&hcrc, OtaBoot_PageBuffer,
                OTA_BOOT_PAGE_SIZE / 4U) != metadata->Page[page].Page_CRC) {
            return 0U;
        }

        MyFLASH_ErasePage(target_address);
        MyFLASH_ProgramBuffer(target_address, OtaBoot_PageBuffer,
            OTA_BOOT_PAGE_SIZE / 4U);

        if (OtaBoot_PageCrc(target_address) != metadata->Page[page].Page_CRC) {
            return 0U;
        }
        HAL_IWDG_Refresh(&hiwdg);
    }
    return 1U;
}
```

```c
static OtaBoot_Result OtaBoot_ActivateSlot(OtaBoot_Slot source_slot)
{
    OtaBoot_Metadata metadata;

    if (!OtaBoot_ValidateSlot(source_slot, &metadata)) {
        return OTA_BOOT_RESULT_ERROR;
    }
    if (!OtaBoot_CopyProgramPages(source_slot, &metadata)) {

            OTA_BOOT_STATUS_RUN_PENDING_CONFIRM)) {
        return OTA_BOOT_RESULT_ERROR;
    }
    return OTA_BOOT_RESULT_ACTIVATED;
}
```

### 7.7 选择 `0x05` 激活候选

```c
OtaBoot_Result OtaBoot_ActivatePendingImage(void)
{
    uint8_t index;
    uint8_t found = 0xFFU;
    OtaBoot_Metadata candidate;
    OtaBoot_Metadata selected;

    for (index = OTA_BOOT_SLOT_2; index <= OTA_BOOT_SLOT_3; index++) {
        if (!OtaBoot_ValidateSlot(index, &candidate)) {
            continue;
        }
        if (candidate.Run_status != OTA_BOOT_STATUS_VALID_BACKUP) {
            continue;
        }
        if ((found == 0xFFU) || OtaBoot_IsNewerId(candidate.ID, selected.ID)) {
            found = index;
            selected = candidate;
        }
    }

    if (found == 0xFFU) {
        return OTA_BOOT_RESULT_SAFE_MODE;
    }
    return OtaBoot_ActivateSlot(found);
}
```

### 7.8 查找当前版本的匹配源槽

```c
static uint8_t OtaBoot_FindMatchingSource(
    const OtaBoot_Metadata* running,
    OtaBoot_Slot* source_slot)
{
    uint8_t slot;
    uint8_t page;

    for (slot = OTA_BOOT_SLOT_2; slot <= OTA_BOOT_SLOT_3; slot++) {
        OtaBoot_Metadata candidate;
        if (!OtaBoot_ReadMetadata(slot, &candidate)) {
            continue;
        }
        if (candidate.ID != running->ID) {
            continue;
        }
        if ((candidate.Run_status != OTA_BOOT_STATUS_RUN_PENDING_CONFIRM) &&
            (candidate.Run_status != OTA_BOOT_STATUS_RUN_CONFIRMED_OK)) {
            continue;
        }
        for (page = 0U; page < OTA_BOOT_PROGRAM_PAGE_COUNT; page++) {
            if (candidate.Page[page].Page_CRC != running->Page[page].Page_CRC) {
                break;
            }
        }
        if (page == OTA_BOOT_PROGRAM_PAGE_COUNT) {
            *source_slot = slot;
            return 1U;
        }
    }
    return 0U;
}
```

### 7.9 IWDG 判坏与回滚

```c
OtaBoot_Result OtaBoot_HandleWatchdogReset(void)
{
    OtaBoot_Metadata running;
    OtaBoot_Slot source_slot;

    if (OtaBoot_ValidateControlPage(OTA_BOOT_SLOT_1, &running)) {
        (void)OtaBoot_WriteMetadata(OTA_BOOT_SLOT_1, &running,
            OTA_BOOT_STATUS_RUN_CONFIRMED_BAD);

        if (OtaBoot_FindMatchingSource(&running, &source_slot)) {
            OtaBoot_Metadata source;
            if (OtaBoot_ReadMetadata(source_slot, &source)) {
                (void)OtaBoot_WriteMetadata(source_slot, &source,
                    OTA_BOOT_STATUS_RUN_CONFIRMED_BAD);
            }
        }
    }

    return OtaBoot_RollbackFromConfirmedBad();
}
```

回滚选择：

```c
OtaBoot_Result OtaBoot_RollbackFromConfirmedBad(void)
{
    uint8_t slot;
    uint8_t found = 0xFFU;
    OtaBoot_Metadata candidate;
    OtaBoot_Metadata selected;

    for (slot = OTA_BOOT_SLOT_2; slot <= OTA_BOOT_SLOT_3; slot++) {
        if (!OtaBoot_ValidateSlot(slot, &candidate)) {
            continue;
        }
        if (candidate.Run_status != OTA_BOOT_STATUS_RUN_CONFIRMED_OK) {
            continue;
        }
        if ((found == 0xFFU) || OtaBoot_IsNewerId(candidate.ID, selected.ID)) {
            found = slot;
            selected = candidate;
        }
    }

    if (found == 0xFFU) {
        return OTA_BOOT_RESULT_SAFE_MODE;
    }

    if (OtaBoot_ActivateSlot(found) == OTA_BOOT_RESULT_ACTIVATED) {
        return OTA_BOOT_RESULT_ROLLED_BACK;
    }
    return OTA_BOOT_RESULT_ERROR;
}
```

### 7.10 正常启动判定

```c
OtaBoot_Result OtaBoot_EvaluateBootState(void)
{
    OtaBoot_Metadata running;
    uint8_t slot;
    uint8_t has_pending = 0U;

    for (slot = OTA_BOOT_SLOT_2; slot <= OTA_BOOT_SLOT_3; slot++) {
        OtaBoot_Metadata candidate;
        if (OtaBoot_ValidateSlot(slot, &candidate) &&
            (candidate.Run_status == OTA_BOOT_STATUS_VALID_BACKUP)) {
            has_pending = 1U;
            break;
        }
    }
    if (has_pending) {
        return OtaBoot_ActivatePendingImage();
    }

    if (!OtaBoot_ValidateSlot(OTA_BOOT_SLOT_1, &running)) {
        return OtaBoot_RollbackFromConfirmedBad();
    }

    if ((running.Run_status == OTA_BOOT_STATUS_RUNNING) ||
        (running.Run_status == OTA_BOOT_STATUS_RUN_PENDING_CONFIRM)) {
        return OTA_BOOT_RESULT_JUMP_RUNNING;
    }

    if (running.Run_status == OTA_BOOT_STATUS_RUN_CONFIRMED_BAD) {
        return OtaBoot_RollbackFromConfirmedBad();
    }

    return OTA_BOOT_RESULT_SAFE_MODE;
}
```

正式实现需要处理“最新 `0x05` 校验失败但另一个 `0x05` 有效”的情况：应跳过坏候选并尝试下一候选，而不是直接进入安全模式。

### 10.3 必测场景

| 场景 | 预期 |
| --- | --- |
| OTA 成功，未复位 | 源槽 `0x05`，一号槽保持原状态 |
| OTA 成功后手动复位 | 一/源槽 `0x08`，跳转新 APP |
| 新 APP 自检成功 | 一号槽 `0x06`，源槽 `0x09`，另一正常旧槽 `0x00` |
| 新 APP 自检失败 | 一/源槽 `0x0A`，触发回滚 |
| 任意 IWDG 复位 | 当前一/源槽 `0x0A`，立即选择另一 `0x09` |
| 回滚 APP 自检成功 | 一号槽 `0x06`，回滚源槽 `0x09`，原坏槽保持 `0x0A` |
| 无回滚版本 | 安全模式，持续喂狗 |
| 复制中掉电 | 下次复位重新验证/复制，不跳半成品 |
| 下一次 OTA 选中坏槽 | 允许覆盖 `0x0A`，成功后重新进入 `0x05` 流程 |

## 11. 关键实现审查点

- `FlashManage_ConfirmRunning()` 只能在最终业务自检通过后调用，不能因“程序进入 main 循环”自动调用。
- `FlashManage_RejectRunning()` 写状态后要确保复位前 Flash 已锁定、写入完成。
- BootLoader 的 IWDG 分支必须先读取原始一号槽元数据，再写 `0x0A`，否则匹配源槽的 ID/CRC 可能丢失。
- 一号槽控制页和源槽控制页必须分别计算自身 Page16 CRC。
- 写入一号槽和源槽状态时必须分别使用各自槽的实际绝对地址，不能用一号槽地址覆盖源槽元数据。
- 所有 Flash 写入函数应返回成功/失败；BootLoader 不得在写入失败后仍跳转 APP。
